<script>
const albums = [
{
name: "Yandhi",
cover: "cover.jpg",
songs: [
{ file: "song1.mp3", title: "Chakras", artist: "Kanye West" },
{ file: "song2.mp3", title: "Calm Before The Storm", artist: "Kanye West (feat. Ty Dolla $ign)" },
{ file: "song3.mp3", title: "The Storm", artist: "Kanye West (feat. Ty Dolla $ign, Kid Cudi,
XXXTENTACION)" },
{ file: "song4.mp3", title: "Spread Your Wings", artist: "Kanye West" }
]
},
{
name: "Second Album",
cover: "cover2.jpg",
songs: [
{ file: "song5.mp3", title: "Song A", artist: "Artist A" },
{ file: "song6.mp3", title: "Song B", artist: "Artist B" }
]
}
];
let currentAlbum = null;
let currentSong = 0;
const library = document.getElementById("library");
const player = document.getElementById("player");
const audio = document.getElementById("audio");
const nowPlaying = document.getElementById("nowPlaying");
// BUILD LIBRARY
albums.forEach((album, i) => {
const div = document.createElement("div");
div.className = "album";
div.innerHTML = `
<img src="${album.cover}">
<div class="album-name">${album.name}</div>
`;
div.onclick = () => openAlbum(i);
library.appendChild(div);
});
// OPEN ALBUM
function openAlbum(i) {
currentAlbum = albums[i];
library.style.display = "none";
player.style.display = "block";
document.getElementById("albumTitle").innerText = currentAlbum.name;
document.getElementById("cover").src = currentAlbum.cover;
buildSongs();
playSong(0);
}
// SONG LIST
function buildSongs() {
const songsDiv = document.getElementById("songs");
songsDiv.innerHTML = "";
currentAlbum.songs.forEach((song, i) => {
const div = document.createElement("div");
div.className = "song";
div.innerText = song.title;
div.onclick = () => playSong(i);
songsDiv.appendChild(div);
});
}
// PLAY SONG
function playSong(i) {
currentSong = i;
const song = currentAlbum.songs[i];
audio.src = song.file;
document.getElementById("title").innerText = song.title;
document.getElementById("artist").innerText = song.artist;
audio.play();
nowPlaying.innerText = `Now Playing: ${song.title} — ${song.artist}`;
highlight(i);
}
// HIGHLIGHT
function highlight(i) {
const songs = document.querySelectorAll(".song");
songs.forEach(s => s.classList.remove("active"));
if (songs[i]) songs[i].classList.add("active");
}
// BACK
function goBack() {
player.style.display = "none";
library.style.display = "grid";
audio.pause();
nowPlaying.innerText = "Nothing playing";
}
</script>
